(()=>{
const R=window.SR;
if(!R||!R.Game)throw new Error('V6.3 base runtime missing');
R.VERSION='0.6.3';
const GP=R.Game.prototype;
const $=s=>document.querySelector(s);
const pick=a=>a&&a.length?a[Math.random()*a.length|0]:null;

// Sanitize every enemy definition before it enters the live simulation.
GP._safeSectorType=function(sector){
  const src=pick((R.ENEMIES||[]).filter(e=>e&&Number.isFinite(+e.hp)&&Number.isFinite(+e.damage)&&Number.isFinite(+e.speed)))||{role:0,hp:1,damage:1,speed:1,shield:0,color:'#8999a5',name:'Unknown Fighter',faction:0};
  const role=Number.isFinite(+src.role)?+src.role:0;
  const t={...src,role,hp:Math.max(.2,+src.hp||1),damage:Math.max(.1,+src.damage||1),speed:Math.max(.15,+src.speed||1),shield:Math.max(0,+src.shield||0),color:sector?.color||src.color||'#8999a5',accent:sector?.accent||src.accent||sector?.color||'#dbe8ef',name:`${sector?.name||'UNKNOWN'} ${R.ROLES?.[role]||'SHIP'}`};
  if(sector?.kind==='pirate'){
    t.pirate=true;t.void=false;t.color='#747b80';t.accent=pick(R.OUTLINE_COLORS)||'#c4d1d8';t.name='Pirate '+(R.ROLES?.[role]||'Raider');
  }else if(sector?.kind==='void'){
    t.void=true;t.pirate=false;t.color='#111318';t.accent=pick(R.OUTLINE_COLORS)||'#8d75ff';t.name='Void '+(['Wisp','Needle','Shade','Maw','Spire','Rift','Harbinger','Abyss'][role%8]||'Form');t.hp*=1.5;t.damage*=1.3;t.speed*=1.07;t.shield=Math.max(t.shield,.28);
  }else{
    t.pirate=false;t.void=false;
    if(sector?.id==='olydran'){t.hp*=1.2;t.speed*=.9;t.shield=Math.max(t.shield,.28)}
    else if(sector?.id==='elsari'){t.shield=Math.max(t.shield,.42);t.speed*=.96}
    else if(sector?.id==='rakkan'){t.damage*=1.18;t.hp*=1.08}
    else if(sector?.id==='svarin'){t.speed*=1.22;t.hp*=.92}
    else if(sector?.id==='ashari'){t.damage*=1.28;t.speed*=1.06}
    else if(sector?.id==='nevari'){t.hp*=1.1;t.shield=Math.max(t.shield,.12)}
    else if(sector?.id==='aaruian'){t.shield=Math.max(t.shield,.34);t.damage*=1.05}
  }
  return t;
};

// Fresh first-wave / sector handoff. No old faction-index assumptions are allowed here.
GP.spawnWave=function(){
  this.wave=(Number(this.wave)||0)+1;
  this.active=true;
  this.paused=false;
  const idx=this.sectorIndex?this.sectorIndex():Math.floor((Math.max(1,this.wave)-1)/10);
  const sector=this.pickSector?this.pickSector(idx):{kind:'base',name:'UNKNOWN',color:'#8999a5',accent:'#dbe8ef'};
  this.currentSector=sector;
  const waveEl=$('#wave');if(waveEl)waveEl.textContent=`◈ ${this.wave} · ${sector.name}`;
  if(this.wave%10===0){this.startSectorBoss();return;}
  const mult=sector.kind==='void'?1.3:1;
  let left=Math.max(1,(6+this.wave*1.85)*mult),guard=0;
  const cap=Math.min(38,10+Math.floor(this.wave/2));
  while(left>0&&(this.en?.length||0)<cap&&guard++<80){
    const type=this._safeSectorType(sector);
    const cost=Math.max(.6,1+(type.role||0)*.55);
    left-=cost;
    this.spawn(type);
  }
  // Absolute fallback. A wave is never allowed to enter an active state with nothing to fight.
  if(!(this.en?.length)){this.spawn(this._safeSectorType(sector));}
};

// Never allow a single runtime exception to permanently kill requestAnimationFrame.
// The first error is surfaced in a tiny diagnostic badge instead of silently freezing the game.
GP._showRuntimeFault=function(err){
  console.error('LoE combat runtime fault',err);
  let n=$('#runtimeFault');
  if(!n){n=document.createElement('div');n.id='runtimeFault';document.body.append(n)}
  n.textContent='COMBAT RECOVERED · '+String(err?.message||err).slice(0,72);
  clearTimeout(this._faultTimer);this._faultTimer=setTimeout(()=>n?.remove(),4500);
};
GP.loop=function(now){
  if(this.dead)return;
  if(!this.last)this.last=now;
  const dt=Math.max(0,Math.min(.04,(now-this.last)/1000||.016));
  this.last=now;
  try{this.update(dt)}catch(e){this._showRuntimeFault(e);this.paused=false;if(!this.active&&!this.bossState)this.count=Math.min(Number(this.count)||.35,.35)}
  try{this.draw()}catch(e){this._showRuntimeFault(e)}
  if(!this.dead)requestAnimationFrame(this.loop);
};

// AUTO belongs to combat only and is always a single control.
const priorBuild63=GP.build;
GP.build=function(){document.querySelectorAll('#autoRun,#runtimeFault').forEach(n=>n.remove());priorBuild63.call(this);const autos=[...document.querySelectorAll('#autoRun')];if(autos.length>1)autos.slice(0,-1).forEach(n=>n.remove())};
})();
