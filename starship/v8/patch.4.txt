      for(let j=i+1;j<rocks.length;j++){const b=rocks[j],dx2=b.x-a.x,dy2=b.y-a.y,d2=Math.hypot(dx2,dy2)||1,min2=a.r+b.r+2;if(d2<min2){const nx=dx2/d2,ny=dy2/d2,over=min2-d2;a.x-=nx*over*.5;a.y-=ny*over*.5;b.x+=nx*over*.5;b.y+=ny*over*.5;const rel=(b.vx-a.vx)*nx+(b.vy-a.vy)*ny;if(rel<0){const imp=-rel*.88;a.vx-=imp*nx;a.vy-=imp*ny;b.vx+=imp*nx;b.vy+=imp*ny}}}
    }
    if(rocks.length===0)this._v8FinishEncounter();
  }else if(e.type==='boss'){
    this.active=true;this.count=999;
    if(e.victoryResolved&&e.traderResolved)this._v8FinishEncounter();
  }
};
