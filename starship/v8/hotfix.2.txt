if(typeof GP.makeBoss!=='function')GP.makeBoss=function(type){
  const list=this.en||(this.en=[]),before=new Set(list);let boss=null;
  try{const made=this.spawn?.(type,true,this.w/2,-110);if(made&&typeof made==='object')boss=made;}catch(err){console.warn('boss spawn fallback',err)}

  if(boss){const i=list.indexOf(boss);if(i>=0)list.splice(i,1)}
  if(!boss){
    const hp=Math.max(1,Number(type.hp)||28),shield=Math.max(0,Number(type.shield)||0);
    boss={type:{...type},boss:true,x:this.w/2,y:-110,vx:0,vy:0,hp,max:hp,shield,maxShield:shield,damage:Number(type.damage)||2.6,speed:.55,size:64,cd:1,phase:1,dead:false,subsystems:[]};

  boss.subsystems=Array.isArray(boss.subsystems)?boss.subsystems:[];

// ---- Denser waves, deliberately staggered ----
GP._v8BeginCombat=function(wave,sector){
  const groups=Math.min(7,3+Math.floor(wave/7));
  const base=Math.min(6,3+Math.floor(wave/12));
  const queue=Array.from({length:groups},(_,i)=>Math.max(2,base+((wave+i)%2)));
  this._v8Encounter={type:'combat',wave,sector,queue,groupsSpawned:0,queueDone:false,complete:false,nextSpawn:0,aliveCap:Math.min(16,7+Math.floor(wave/6))};
  this.en=[];this.rocks=[];this.bad=[];this.tele=[];this.active=true;this.count=999;this._v8SpawnGroup(true);
};
GP._v8SpawnGroup=function(force=false){
  const e=this._v8Encounter;if(!e||e.type!=='combat'||!e.queue?.length)return;
  if(!force&&(e.nextSpawn||0)>0)return;
  const n=e.queue.shift();for(let i=0;i<n;i++){const type=this._v8SpawnType?.(e.sector);if(type)this.spawn(type)}
  e.groupsSpawned=(e.groupsSpawned||0)+1;e.queueDone=e.queue.length===0;e.nextSpawn=2.5+Math.random()*1.5;this.active=true;this.count=999;
};
const pacedUpdate=GP.update;
GP.update=function(dt){
  pacedUpdate.call(this,dt);const e=this._v8Encounter;if(this.dead||this.paused||!e||e.type!=='combat'||e.complete||e.queueDone)return;
  e.nextSpawn=Math.max(0,(e.nextSpawn||0)-dt);
  if(e.nextSpawn<=0&&(this.en?.length||0)<(e.aliveCap||10))this._v8SpawnGroup(true);
};

}
